{
  "15 levels": (() => {
    // A string rendering stops descending at its depth limit, and whatever
    // lies below is elided. The structured conversion's own default depth is
    // deeper, so its rendering carries the whole value.
    let value = "bottom";
    for (let i = 0; i < 15; i++) {
      value = { d: value };
    }
    return value;
  })(),
}
