{
  // A hole renders as `void`, and a run of holes as `void` times the length
  // of the run.
  "with holes": [1, , 3, , , 6, , ],
  "holes across the limit": (() => {
    // A run of holes which crosses the default limit of 100 renders as only
    // the holes below the limit.
    const a = [1];
    a.length = 250;
    return a;
  })(),
}

with holes:
[1,void,3,void*2,6,void]
[
  1,
  void,
  3,
  void * 2,
  6,
  void
]

holes across the limit:
[1,void*99,...length:250]
[
  1,
  void * 99,
  ... length: 250
]
