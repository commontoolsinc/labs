const [directory, output] = Deno.args;
const root = new URL(`file://${directory.replace(/\/$/, "")}/`);
const variants = ["recorded-source/seed-check.ts.txt", "seed-check.ts.txt"];
const AsyncFunction = Object.getPrototypeOf(async function () {}).constructor;
const rows = [];
for (const file of variants) {
  const bytes = await Deno.readFile(new URL(file, root));
  const source = new TextDecoder().decode(bytes);
  const hash = Array.from(new Uint8Array(await crypto.subtle.digest("SHA-256", bytes)), (b) => b.toString(16).padStart(2, "0")).join("");
  const countStart = source.indexOf("const topicCount =");
  const countCode = countStart < 0
    ? 'return Number(Deno.env.get("CF_TOPIC_BOARD_TOPICS") ?? 5);'
    : source.slice(countStart, source.indexOf("const identity =", countStart)) + "return topicCount;";
  for (const configured of ["0", "-1", "1.5", "NaN", "", "5"]) {
    let result, error;
    try { result = await new AsyncFunction("Deno", countCode)({env:{get:()=>configured}}); }
    catch (e) { error = e instanceof Error ? e.message : String(e); }
    rows.push({file,hash,case:"topicCount",configured,result,error});
    if (countStart >= 0 && ((configured === "5") !== (error === undefined))) throw new Error(`Wrong validation for ${configured}`);
  }
  const afterFixture = source.indexOf('JSON.stringify(fixture));') + 'JSON.stringify(fixture));'.length;
  const statsCode = source.slice(afterFixture, source.indexOf("const reader =", afterFixture));
  for (const status of [200, 503]) {
    const writes: unknown[] = [];
    let error;
    try {
      await new AsyncFunction("Deno", "fetch", "apiUrl", "runDir", "join", statsCode)(
        {writeTextFile:(...args: unknown[])=>writes.push(args)},
        () => Promise.resolve({ok:status===200,status,text:()=>Promise.resolve(JSON.stringify({status}))}),
        "http://synthetic.invalid", "capture", (...parts: string[])=>parts.join("/"),
      );
    } catch (e) { error = e instanceof Error ? e.message : String(e); }
    rows.push({file,hash,case:"stats",status,writes,error});
    if (countStart >= 0 && (status === 503 ? error === undefined || writes.length !== 0 : error !== undefined || writes.length !== 1)) throw new Error(`Wrong statistics validation for ${status}`);
  }
}
await Deno.writeTextFile(output, JSON.stringify({mode:"source-fragment input controls, no network or pattern workload",rows}, null, 2)+"\n");
console.log("Replay input controls passed.");
